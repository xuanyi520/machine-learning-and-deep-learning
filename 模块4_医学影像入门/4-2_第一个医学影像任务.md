## 一、完整流程（工业界也是这七步）

```
1. 明确问题      →  是分类还是分割？要什么指标？
2. 找数据        →  公开数据集或导师的临床数据
3. 数据探索 EDA  →  样本量、类别分布、图像长什么样
4. 预处理        →  尺寸统一、归一化、增强
5. 划分数据      →  训练/验证/测试
6. 建模训练      →  网络、损失、优化器
7. 评估与报告    →  指标、混淆矩阵、错误分析
```

**新手最容易跳过的第 3 步和第 7 步，恰恰是医学 AI 最看重的。**

临床合作者不关心你的网络有几层，他们关心：**你的模型漏诊了几个？错在哪类病人上？**

## 四、第 4-6 步：预处理 + 建模

```python
from torchvision import transforms
from torch.utils.data import DataLoader
import torch.nn as nn
import torch.optim as optim

# ===== 预处理 =====
train_transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.5], std=[0.5]),      # 单通道
])
eval_transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.5], std=[0.5]),
])

train_set = DataClass(split='train', transform=train_transform, download=True)
val_set   = DataClass(split='val',   transform=eval_transform, download=True)
test_set  = DataClass(split='test',  transform=eval_transform, download=True)

train_loader = DataLoader(train_set, batch_size=64, shuffle=True)
val_loader   = DataLoader(val_set,   batch_size=64, shuffle=False)
test_loader  = DataLoader(test_set,  batch_size=64, shuffle=False)

# ===== 模型（复用 3-4 的结构）=====
n_classes = len(info['label'])

class MedCNN(nn.Module):
    def __init__(self, in_ch, n_cls):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(in_ch, 32, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
            nn.Conv2d(32, 64, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(64 * 7 * 7, 64), nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(64, n_cls),
        )

    def forward(self, x):
        return self.classifier(self.features(x))

device = "cuda" if torch.cuda.is_available() else "cpu"
model = MedCNN(info['n_channels'], n_classes).to(device)
print("参数量:", sum(p.numel() for p in model.parameters()))

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=1e-3)

# ===== 训练（带验证集监控）=====
def run_epoch(model, loader, train=True):
    model.train() if train else model.eval()
    total_loss, correct, total = 0, 0, 0

    ctx = torch.enable_grad() if train else torch.no_grad()
    with ctx:
        for xb, yb in loader:
            xb = xb.to(device)
            yb = yb.squeeze().long().to(device)

            if train:
                optimizer.zero_grad()
            out = model(xb)
            loss = criterion(out, yb)
            if train:
                loss.backward()
                optimizer.step()

            total_loss += loss.item() * xb.size(0)
            correct += (out.argmax(1) == yb).sum().item()
            total += xb.size(0)

    return total_loss / total, correct / total

history = {'train_loss': [], 'val_loss': [], 'train_acc': [], 'val_acc': []}

for epoch in range(10):
    tr_loss, tr_acc = run_epoch(model, train_loader, train=True)
    va_loss, va_acc = run_epoch(model, val_loader, train=False)

    history['train_loss'].append(tr_loss)
    history['val_loss'].append(va_loss)
    history['train_acc'].append(tr_acc)
    history['val_acc'].append(va_acc)

    print(f"Epoch {epoch+1:2d}  "
          f"train loss={tr_loss:.4f} acc={tr_acc:.4f} | "
          f"val loss={va_loss:.4f} acc={va_acc:.4f}")
```

**注意这里用了验证集。** 前面的模块你都只用训练/测试，真实项目必须有验证集——否则你没法判断什么时候该停止训练。

## 五、第 7 步：评估与报告（本模块重点）

### 5.1 画训练曲线，判断过拟合

```python
fig, axes = plt.subplots(1, 2, figsize=(12, 4))

axes[0].plot(history['train_loss'], label='训练')
axes[0].plot(history['val_loss'], label='验证')
axes[0].set_xlabel('Epoch'); axes[0].set_ylabel('Loss')
axes[0].set_title('损失曲线'); axes[0].legend()

axes[1].plot(history['train_acc'], label='训练')
axes[1].plot(history['val_acc'], label='验证')
axes[1].set_xlabel('Epoch'); axes[1].set_ylabel('准确率')
axes[1].set_title('准确率曲线'); axes[1].legend()

plt.tight_layout(); plt.show()
```

**判读方法：**

- 两条线都还在降 → 可以继续训练
- 训练降、验证升 → 过拟合，该停了
- 两条线离得很远 → 过拟合严重
- 训练都不降 → 欠拟合，模型太小或学习率不对

### 5.2 医学指标（不是只看准确率）

```python
from sklearn.metrics import (classification_report, confusion_matrix,
                             roc_auc_score, roc_curve)
import seaborn as sns

model.eval()
all_probs, all_preds, all_labels = [], [], []

with torch.no_grad():
    for xb, yb in test_loader:
        xb = xb.to(device)
        probs = torch.softmax(model(xb), dim=1)
        all_probs.append(probs.cpu().numpy())
        all_preds.append(probs.argmax(1).cpu().numpy())
        all_labels.append(yb.squeeze().numpy())

all_probs = np.concatenate(all_probs)
all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

# 分类报告：精确率/召回率/F1 全都有
target_names = [label_names[str(i)] for i in range(n_classes)]
print(classification_report(all_labels, all_preds, target_names=target_names, digits=4))

# 混淆矩阵
cm = confusion_matrix(all_labels, all_preds)
plt.figure(figsize=(6, 5))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=target_names, yticklabels=target_names)
plt.xlabel('预测'); plt.ylabel('真实'); plt.title('混淆矩阵')
plt.tight_layout(); plt.show()

# AUC（二分类时）
if n_classes == 2:
    auc = roc_auc_score(all_labels, all_probs[:, 1])
    print(f"\nAUC = {auc:.4f}")

    fpr, tpr, _ = roc_curve(all_labels, all_probs[:, 1])
    plt.figure(figsize=(6, 5))
    plt.plot(fpr, tpr, label=f'AUC = {auc:.4f}')
    plt.plot([0, 1], [0, 1], 'k--', label='随机猜测')
    plt.xlabel('假阳性率 FPR'); plt.ylabel('真阳性率 TPR（召回率）')
    plt.title('ROC 曲线'); plt.legend()
    plt.show()
```

**注意 `classification_report` 输出的 `recall` 就是 W7 讲的召回率，也就是灵敏度。** 在医学报告里，**这个值比 accuracy 重要**。

### 5.3 错误分析（最容易被跳过，但最值钱）

```python
# 找出模型判错的样本
wrong_idx = np.where(all_preds != all_labels)[0]
print(f"测试集共 {len(all_labels)} 张，错了 {len(wrong_idx)} 张")

# 画出前 16 个错误样本
n_show = min(16, len(wrong_idx))
if n_show > 0:
    fig, axes = plt.subplots(2, 8, figsize=(16, 4))
    for ax, idx in zip(axes.flat, wrong_idx[:n_show]):
        img, _ = test_set[idx]
        ax.imshow(img.squeeze(), cmap='gray')
        ax.set_title(f"预测:{target_names[all_preds[idx]]}\n真实:{target_names[all_labels[idx]]}",
                     fontsize=7, color='red')
        ax.axis('off')
    for ax in axes.flat[n_show:]:
        ax.axis('off')
    plt.suptitle('模型判错的样本')
    plt.tight_layout(); plt.show()
```

**认真看这些错图，问自己：**

- 人眼能分辨吗？
- 是不是某些类别系统性出错？
- 图像质量有问题吗？
- **临床上，这种错误会造成什么后果？**

**错误分析写进报告，比再刷 1% 准确率更有价值。**
