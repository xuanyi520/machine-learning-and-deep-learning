```python
# 文件：ans_13ab.py
import pandas as pd
import numpy as np
from scipy import stats

PATH = r'E:\AI学习\练习代码\data\uci\Hepatitis.csv'
df = pd.read_csv(PATH, na_values=['?'])
df.columns = df.columns.str.strip().str.replace(' ', '_')
df['death'] = (df['Class'] == 1).astype(int)
df['male'] = df['Sex'].map({1: 1, 2: 0})
df['ascites'] = df['Ascites'].map({1: 1, 2: 0})
df['spiders'] = df['Spiders'].map({1: 1, 2: 0})
df['varices'] = df['Varices'].map({1: 1, 2: 0})
df['liver_firm'] = df['Liver_Firm'].map({2: 1, 1: 0})
df['steroid'] = df['Steroid'].map({2: 1, 1: 0})

def pick_test(data, var, group):
    """按变量类型和前提自动选检验。"""
    sub = data[[var, group]].dropna()
    g1 = sub.loc[sub[group] == 1, var]
    g0 = sub.loc[sub[group] == 0, var]

    if pd.api.types.is_numeric_dtype(sub[var]) and sub[var].nunique() > 2:
        sw1 = stats.shapiro(g1).pvalue
        sw0 = stats.shapiro(g0).pvalue
        if sw1 > 0.05 and sw0 > 0.05:          # 两组都要近似正态
            p = stats.ttest_ind(g1, g0, equal_var=False).pvalue
            return ('mean ± SD', f'{g1.mean():.2f} ± {g1.std():.2f}',
                    f'{g0.mean():.2f} ± {g0.std():.2f}', p, 'Welch t',
                    len(g1), len(g0))
        p = stats.mannwhitneyu(g1, g0, alternative='two-sided').pvalue
        return ('median (IQR)',
                f'{g1.median():.2f} ({g1.quantile(.25):.2f}-{g1.quantile(.75):.2f})',
                f'{g0.median():.2f} ({g0.quantile(.25):.2f}-{g0.quantile(.75):.2f})',
                p, 'Mann-Whitney U', len(g1), len(g0))

    tab = pd.crosstab(sub[var], sub[group])
    _, p_chi, _, exp = stats.chi2_contingency(tab, correction=False)
    if exp.min() < 5:
        p, test = stats.fisher_exact(tab.values)[1], 'Fisher'
    else:
        p, test = p_chi, 'chi-square'
    a = int(((sub[var] == 1) & (sub[group] == 1)).sum())
    b = int(((sub[var] == 1) & (sub[group] == 0)).sum())
    return ('n (%)', f'{a} ({a/len(g1):.1%})', f'{b} ({b/len(g0):.1%})',
            p, test, len(g1), len(g0))

# 逐变量打印 Shapiro p，才能回答"为什么"
for var in ['Age', 'Bilirubin', 'Alk_Phosphate', 'Sgot', 'Albumin', 'Protime']:
    sub = df[[var, 'death']].dropna()
    g1 = sub.loc[sub.death == 1, var]
    g0 = sub.loc[sub.death == 0, var]
    print(f'{var}: n={len(g1)}/{len(g0)}, '
          f'Shapiro p={stats.shapiro(g1).pvalue:.4g}/{stats.shapiro(g0).pvalue:.4g}, '
          f'偏度={stats.skew(g1):.2f}/{stats.skew(g0):.2f}')

# 第 1 问：Sgot -> Protime
# 第 2 问：10 个死亡病人的 Albumin 置缺失
rng = np.random.default_rng(0)
df2 = df.copy()
dead_idx = df2.index[df2.death == 1].to_numpy()
df2.loc[rng.choice(dead_idx, 10, replace=False), 'Albumin'] = np.nan
print(pick_test(df, 'Albumin', 'death'))
print(pick_test(df2, 'Albumin', 'death'))

# 对照：同样的 10 例缺失，作用在边缘显著的碱性磷酸酶上
df3 = df.copy()
df3.loc[rng.choice(dead_idx, 10, replace=False), 'Alk_Phosphate'] = np.nan
print(pick_test(df, 'Alk_Phosphate', 'death'))
print(pick_test(df3, 'Alk_Phosphate', 'death'))
```

**判分要点**

- 第 1 问的结论必须同时包含"方法没变"和"为什么没变"：Protime 在死亡组的 Shapiro p = 0.0009、偏度 1.59，在存活组 p = 0.026、偏度 -0.28，**两组至少一组拒绝正态**，所以仍走 Mann-Whitney。同时要指出真正变的是信息量：Protime 缺失 43.2%，非缺失只剩 18/70 例，p 从 0.223 变成 2.7e-05 是"信号强 + 样本少"共同作用，不是检验方法升级了。
- 第 2 问必须报告"缺失前 n=27/112、缺失后 n=21/112"和两次 p：**1.733e-08 → 2.552e-08**。要点在于：10 个被点名的人里有 4 个白蛋白本来就是缺失，所以实际只少了 6 例（27→21），不是 10 例。这个细节能区分"真的跑了代码"和"编了个 10"。
- 白蛋白的 p 只变差 1.5 倍，**不能**拿它来证明"功效下降"，因为白蛋白的效应太大（中位数 3.30 对 4.00，Cohen's d 约 -1.46），少 6 例撼动不了结论。要证明"缺失削弱功效"，必须换成边缘信号：**碱性磷酸酶 p 从 0.0344 升到 0.198，跨过 0.05；Protime 的 p 放大约 5 倍**。这才是"统计功效下降，不是效应消失"的直接证据。
- 解释功效下降的机制要落到样本量和标准误：Mann-Whitney 的检验效能取决于两组的有效样本量，死亡组从 24 降到 16（碱性磷酸酶），秩和的标准误变大，同样的中位数差就跨不过阈值。效应量（中位数差、秩相关）几乎没变，变的只是精度。
- 回归审稿人问题的标准答法（第 3 问）："白蛋白在死亡组 Shapiro-Wilk p = 0.471，看似正态；但存活组 p = 7.9e-06、偏度 0.26，且两组样本量 27 对 112 严重不等、方差不齐。t 检验要求两组都近似正态且方差相近，这里两条都不满足。Mann-Whitney 不要求正态，比较的是分布位置，与中位数 (IQR) 的描述方式也匹配。同一结论在 log-rank 和 Cox 里方向一致，不是检验方法造成的。"
- 表 1 的百分比分母必须是该变量的非缺失例数（白蛋白死亡组 27，不是 32），这一点在输出里已经体现。12 个变量做了 12 次检验、未校正，若要在正文声称"基线有差异"，要说明这是描述性比较。

**常见错误**

- 只报"因为数据偏态"而不报 Shapiro p。验收标准明确要求具体数字，至少要给出"白蛋白存活组 p = 7.9e-06"这一条。
- 看到死亡组白蛋白 Shapiro p = 0.47 就说"应该用 t 检验"。**前提是两组都正态**，一组不满足就得换非参数方法。
- 把"缺失 10 例后 p 变大"直接解释成"效应消失了"。p 变大可能只是样本少了，必须同时看中位数差是否还在（本例 3.30→3.10，效应依然很大）。
- 用 `.astype(int)` 把 NaN 变成 0。`df['ascites'] = (df['Ascites'] == 1).astype(int)` 会把缺失当成"无腹水"，阳性率被稀释。必须用 `map({1:1, 2:0})` 保留 NaN。
- 照抄 UCI codebook 的编码方向。本文件腹水是 1 = 有（1 组死亡率 45.2%，2 组 5.0%），与文档相反；抗病毒药也是 1 = 使用（1 组死亡率 8.3%，2 组 22.9%）。判断方法就是算两组的结局率，看哪个方向符合临床。
- 分类变量一律用卡方。男性、腹水、静脉曲张的最小期望频数小于 5（女性死亡人数为 0），必须用 Fisher。
- 认为"Fisher 只适合 n < 30"。判断标准是期望频数，不是总样本量。
- 表格里用均值 ± 标准误（SE）。表 1 要的是标准差（SD），SE 会随样本量缩小，造成"数据很集中"的假象。
- 缺失处理不交代。删掉 10 例后样本从 27 变 21，表里必须写清新的分母，否则读者会以为还是 32 例死亡。

### 练习 13-B（p 值、OR 和置信区间）

**参考代码**

```python
# 文件：ans_13ab.py（13-B 部分）
import numpy as np
import pandas as pd
from scipy import stats

a, b, c, d = 20, 60, 10, 90

# 1. OR 与 95% CI（Woolf 法）
OR = (a * d) / (b * c)
lnOR = np.log(OR)
SE = np.sqrt(1/a + 1/b + 1/c + 1/d)
lo, hi = np.exp(lnOR - 1.96 * SE), np.exp(lnOR + 1.96 * SE)
print(f'OR = ({a}×{d})/({b}×{c}) = {a*d}/{b*c} = {OR:.4f}')
print(f'ln(OR) = {lnOR:.6f}')
print(f'SE(lnOR) = sqrt(1/{a}+1/{b}+1/{c}+1/{d}) = {SE:.6f}')
print(f'95% CI = {lo:.4f} 到 {hi:.4f}')

# 2. RR
r1, r0 = a / (a + b), c / (c + d)
RR = r1 / r0
lnRR = np.log(RR)
SE_rr = np.sqrt(1/a - 1/(a+b) + 1/c - 1/(c+d))
print(f'RR = {RR:.4f}，95% CI '
      f'{np.exp(lnRR-1.96*SE_rr):.4f}-{np.exp(lnRR+1.96*SE_rr):.4f}')

# 3. 卡方与 Fisher
tab = np.array([[a, b], [c, d]])
chi2, p_chi, _, exp = stats.chi2_contingency(tab, correction=False)
print(f'卡方 chi2={chi2:.4f}, p={p_chi:.4f}, 最小期望={exp.min():.2f}')
print(f'Fisher p={stats.fisher_exact(tab)[1]:.4f}')

# 4. 构造数据，用 fit_logit 验证（Newton-Raphson，与 statsmodels 一致）
sim = pd.DataFrame({'exposed': np.r_[np.ones(a + b), np.zeros(c + d)],
                    'death': np.r_[np.ones(a), np.zeros(b),
                                   np.ones(c), np.zeros(d)]})

def fit_logit(X, y, names):
    X = np.asarray(X, float)
    y = np.asarray(y, float)
    X1 = np.column_stack([np.ones(len(X)), X])
    beta = np.zeros(X1.shape[1])
    for _ in range(50):
        p = 1 / (1 + np.exp(-(X1 @ beta)))
        W = p * (1 - p) + 1e-12
        grad = X1.T @ (y - p)
        info = X1.T @ (X1 * W[:, None])
        step = np.linalg.solve(info, grad)
        beta = beta + step
        if np.max(np.abs(step)) < 1e-9:
            break
    p = 1 / (1 + np.exp(-(X1 @ beta)))
    W = p * (1 - p)
    cov = np.linalg.inv(X1.T @ (X1 * W[:, None]))
    se = np.sqrt(np.diag(cov))
    out = []
    for i, name in enumerate(names, start=1):
        z = beta[i] / se[i]
        out.append([name, np.exp(beta[i]),
                    np.exp(beta[i] - 1.96 * se[i]),
                    np.exp(beta[i] + 1.96 * se[i]),
                    2 * stats.norm.sf(abs(z))])
    return pd.DataFrame(out, columns=['变量', 'OR', '95% CI 下限',
                                      '95% CI 上限', 'p'])

print(fit_logit(sim[['exposed']], sim['death'], ['exposed']))
```

**判分要点**

- 手算过程必须完整写出四步：OR = 3.00 → ln(OR) = 1.0986 → SE = 0.4216 → CI = exp(1.0986 ± 1.96×0.4216) = **1.3129 到 6.8553**。验收线是 OR = 3.00、CI 1.31-6.86，误差小于 0.01。
- RR = 2.50，且要给出 RR 的 CI：SE(lnRR) = sqrt(1/20 - 1/80 + 1/10 - 1/100) = 0.3571，95% CI **1.2416-5.0337**。只报 RR 点估计不够，审稿人会要 CI。
- 必须解释 OR 为什么比 RR 大：结局不罕见（暴露组 25%、对照组 10%，合并约 16.7%）。OR 的分母是"非事件"的 odds，当事件率上升时，odds 的增长快于 risk，所以 OR 会放大 RR。本例放大 **(3.00-2.50)/2.50 = 20%**。正确写法是"odds 增加 3 倍"，不是"风险增加 3 倍"。
- 代码验证要与手算完全一致：`fit_logit` 返回 OR 3.0000、CI 1.3129-6.8553，与手算差 0.000000。logistic 回归的系数就是 ln(OR)，这是"2x2 表"和"回归"之间的桥梁，必须点明。
- 卡方 7.20（p = 0.0073）与 Fisher 0.0089 的差别要能解释：2x2 表最小期望频数 13.33，远大于 5，理论上该用不校正卡方；Fisher 更保守，p 略大。两者结论一致（都显著）。Yates 校正后 p = 0.0131，也显著，但 Yates 校正对 2x2 偏保守，现代做法是期望频数够就报不校正卡方。
- 论文结果句（可直接用）：

  > 暴露组死亡 20/80（25.0%），非暴露组死亡 10/100（10.0%）。暴露与死亡风险升高相关（OR = 3.00, 95% CI 1.31-6.86, p = 0.009；Fisher 精确检验 p = 0.009）。由于结局发生率约 17%，OR 会高估风险比，暴露组的绝对风险增加为 15.0 个百分点（25.0% vs 10.0%），相应的 RR 为 2.50（95% CI 1.24-5.03）。

  注意句中用的是"odds"和"绝对风险差"，没有把 OR 说成"风险倍数"。

- 样本量的局限要提一句：每格只有 10 到 20 人，CI 从 1.31 到 6.86 很宽，点估计 3.0 的不确定性很大。

**常见错误**

- 把 OR = 3.0 写成"风险增加 3 倍"。这是本题最核心的考点。风险从 10% 到 25% 是增加 1.5 倍（RR = 2.5），odds 才是 3 倍。
- SE 公式用错：写成 sqrt(1/20 + 1/60 + 1/10 + 1/90) 之外的形式，比如用 1/(a+b) 代替 1/a。四项分别是四个格子的倒数，一个都不能少。
- CI 在 OR 尺度上做加减：`OR ± 1.96*SE`。必须在 ln 尺度上算完再 exp 回来。
- 用 `exp(1.96*SE)` 当倍数直接乘 OR，写成 `3.0 * 2.26`。数学上等价于上界，但下界会算错；而且丢了 ln 尺度就没法讲清原理。
- 报 RR 却不报 CI，或 RR 的 SE 用了 OR 的公式。RR 的 SE 是 sqrt(1/a - 1/n1 + 1/c - 1/n0)，注意是减号。
- 用 Yates 校正的卡方作为主结果却不说明。2x2 且期望频数充足时，惯例报不校正卡方；要报 Yates 就得写清。
- 把 p = 0.0092 写成 p = 0.000。软件输出的小 p 值要写成 p < 0.001。
- 用 t 检验比较两个比例，或用卡方检验连续变量。检验方法和数据类型必须匹配。
- 忽略"这是回顾性 2x2 表，OR 只是关联". 没有做混杂调整前，不能说"暴露导致死亡"。

### 练习 13-C（多重检验校正）

**参考代码**

```python
# 文件：ans_13cd.py（13-C 部分）
import numpy as np
import pandas as pd
from scipy import stats

df = pd.read_csv(r'E:\AI学习\练习代码\data\uci\Hepatitis.csv', na_values=['?'])
df.columns = df.columns.str.strip().str.replace(' ', '_')
df['death'] = (df['Class'] == 1).astype(int)

def p_value(var):
    sub = df[[var, 'death']].dropna()
    if sub[var].nunique() > 2:
        a = sub.loc[sub['death'] == 1, var]
        b = sub.loc[sub['death'] == 0, var]
        return stats.mannwhitneyu(a, b, alternative='two-sided').pvalue
    tab = pd.crosstab(sub[var], sub['death']).values
    return stats.fisher_exact(tab)[1]

base10 = ['Age', 'Spleen_Palpable', 'Alk_Phosphate', 'Sex', 'Steroid',
          'Anorexia', 'Antivirals', 'Sgot', 'Liver_Firm', 'Liver_Big']
extra5 = ['Bilirubin', 'Albumin', 'Protime', 'Ascites', 'Varices']

def bh(raw):
    """Benjamini-Hochberg：q = p*m/rank，并从大到小保证单调。"""
    m = len(raw)
    order = np.argsort(raw)
    q = np.empty(m)
    prev = 1.0
    for rank in range(m, 0, -1):
        idx = order[rank - 1]
        prev = min(prev, raw[idx] * m / rank)
        q[idx] = prev
    return q

def report(names, tag):
    raw = np.array([p_value(v) for v in names])
    q = bh(raw)
    bonf = np.minimum(raw * len(raw), 1.0)
    out = pd.DataFrame({'指标': names, '原始 p': raw, 'BH q': q,
                        'Bonferroni p': bonf}).sort_values('原始 p')
    print(out.to_string(index=False))
    print(f'原始 p<0.05: {(raw<0.05).sum()} 个 | BH: {(q<0.05).sum()} 个 | '
          f'Bonferroni: {(bonf<0.05).sum()} 个 | 阈值 {0.05/len(names):.4f}')
    return out

report(base10, '10 个')
report(base10 + extra5, '15 个')
```

**判分要点**

- 三个计数必须报全：10 个指标时原始显著 **4** 个、BH 后 **2** 个、Bonferroni 后 **1** 个；15 个指标时原始 **9** 个、BH 后 **7** 个、Bonferroni 后 **6** 个。阈值分别写明 0.05/10 = 0.0050、0.05/15 = 0.0033。
- q 值的定义必须准确：**q 不是"该指标为真的概率"，也不是"假阳性概率"，而是"被判为阳性的这一批里，假阳性的期望比例"**。写成"q = 0.024 表示这个发现只有 2.4% 是假的"是错的；正确说法是"如果把这批 q < 0.05 的都判为阳性，其中假阳性约占 5%"。
- 第 2 问是本练习的核心：加入 5 个强信号后，边缘指标的 q **反而变小**（碱性磷酸酶 0.11 → 0.0645，性别 0.11 → 0.0733）。原因在 BH 的公式 q(k) = p(k) × m / k：m 从 10 增到 15（惩罚变重），但这 5 个新指标全部排在前面（k = 1 到 5），把边缘指标的**排名 k 从第 3、第 4 推到了第 8、第 9**。k 变大意味着分母变大，同时强信号把 q 的单调化链条往下压。一句话：**BH 按排名分配惩罚，前排的强信号会"掩护"后排的边缘信号。**
- 但要注意：q 变小不等于"变显著"。碱性磷酸酶 q = 0.0645 仍大于 0.05，只是从"远不显著"变成"接近显著"。这个细节能区分背公式和真理解。
- 第 3 问：如果这 15 个指标**全部是预先设定的主要终点**，仍然需要校正，因为"多个主要终点"本身就是一个 family，不校正会让 FWER 膨胀到 1 - 0.95^15 = 53.7%。正确的设计是：**主要终点只设一个**，其余降为次要终点并明确校正策略。反过来说，如果 15 个都是探索性筛选，用 BH 控制 FDR 是合适的；如果其中某一个事先被定为唯一主要终点，那一个不需要校正（α = 0.05），其余 14 个另成一个 family。
- Bonferroni 在指标相关时过保守要能举例：年龄、胆红素、白蛋白、凝血酶原时间彼此相关（都是肝功能/门脉高压的代理），Bonferroni 假设 15 次检验独立，除以 15 相当于把相关指标的惩罚重复计算，会连真信号一起扔（本例 Bonferroni 只剩 6 个，而 BH 保留 7 个，脾可触及被 Bonferroni 误杀）。
- 要说清"校正方法必须预先设定"。看完数据再决定用 BH 还是 Bonferroni，等于用方法选择来换显著性，是另一种 p-hacking。

**常见错误**

- 把 q 值叫成"校正后的 p 值"并当 p 解释。可以叫 adjusted p，但解释必须是 FDR 的语言，不能是"原假设为真的概率"。
- BH 实现时忘记单调化（从大到小取 min），导致排名靠后的 q 反而比前面小。必须做 `prev = min(prev, ...)`。
- 只报"校正后剩几个"，不报原始 p 和 q 的具体数值，读者无法复核。
- 认为"加了强信号，多重比较的惩罚变重，所有 q 都应该变大"。这是只看到 m 变大、没看到 rank 变化。本题正是反例。
- 把 15 个指标塞进一个 family 后，又对其中某一个单独宣称"这是主要终点所以不校正"，属于事后指定主要终点。
- 用 Bonferroni 处理组学数据或高度相关的量表条目，导致全部被筛掉，然后结论写"没有发现任何差异"。
- 把"校正后不显著"写成"该指标无效"。应写"校正后未达显著，可能受样本量和指标相关性限制，列为待验证候选"。
- 忘记报告检验类型。15 个指标里连续变量用 Mann-Whitney、分类变量用 Fisher，方法不统一会让 p 值的可比性受质疑。

- KM 图不画 95% CI、不报 number at risk，尾部只剩 4 个人还敢解读。
- 中位生存时间报成"平均生存时间"。中位数是曲线与 0.5 的交点；曲线没降到 0.5 时应报"未达到"。

- 分段 Cox 的 time zero 处理错：第二段必须把时间重置为 T - 12，否则同一批人被算了两次绝对时间。
- 忘记检查删失是否非信息。如果失访与预后相关（病情重的人失访），KM 和 Cox 都会有偏，要在讨论里说明。
- 用 Schoenfeld 检验时选错时间变换（rank 与 km 结果可能不同），却不说明用了哪一种。报告里要写"rank 时间变换"。

```python
    S = np.cov(v01) / m + np.cov(v10) / n
    var = S[0, 0] + S[1, 1] - 2 * S[0, 1]
    z = (aucs[0] - aucs[1]) / np.sqrt(var)
    return aucs[0], aucs[1], z, 2 * stats.norm.sf(abs(z))

print(delong_test(y, p_full, p_age))

    nb_all.append(prev - (1 - prev) * pt / (1 - pt))

better = (nb_model > nb_all) & (nb_model > 0)
```

**判分要点**

- PPV 必须用**贝叶斯公式重新算**，不能把研究队列的 0.414 直接搬到其他人群。四个患病率的具体结果：

  | 1% | 906 | 33,000 | **2.67%** | 99.86% |
  | 0.1% | 91 | 33,300 | **0.27%** | 99.99% |

  0.1% 那一行是最有说服力的：**每 10 万人里，真阳性 91 人，假阳性 33,300 人；每发现 1 例真病人，要检测约 368 人。** 阳性者中真正有病的只有 0.27%。
- 要解释"灵敏度、特异度不变，PPV 变"的原因：分母里假阳性的数量是 (1 - 特异度) × (1 - 患病率) × N，患病率越低，(1 - 患病率) 越接近 1，假阳性的绝对数量压过真阳性。特异度 0.667 意味着假阳性率 33.3%，这是本模型最致命的短板。
- DeLong 结果要报全三个数：AUC 0.821 vs 0.633，ΔAUC = +0.188，**z = 3.303，p = 0.00096**。结论：6 项指标显著优于仅年龄，年龄单独用没有筛查价值（AUC 0.633 接近随机）。
- DCA 要报阈值区间和具体净收益：模型优于"全部治疗"且优于"不治疗"的区间是 **pt = 0.05 到 0.58**。在 pt = 0.20 时，模型净收益 0.1129，全部治疗只有 0.0081，不治疗为 0；pt = 0.50 时模型仍有 0.0065 的正净收益，而全部治疗是 -0.5871。这说明模型在很宽的阈值范围内都有临床价值。
- 但这个"有价值"只在本研究的患病率（20.65%）下成立。DCA 的净收益公式里含患病率，换人群要重算。这是第 4 问的关键：**AUC 和 DCA 都不能直接外推到社区人群。**
- 第 4 问的标准答案（三条理由，缺一不可）：
  1. **患病率**：社区人群肝癌患病率约 0.1%，本模型 PPV 只有 0.27%，阳性预测值低到无法用于告知或转诊。
  2. **假阳性负担**：每 10 万人产生约 33,300 个假阳性，意味着大量不必要的增强 CT/MRI、穿刺活检和焦虑，医疗系统无法承受。
  3. **特异度不足**：0.667 的特异度在低患病率下是灾难性的；即使把特异度提到 0.90，PPV 也只有 0.89%（讲义 5.3 节），仍不到 1%。要提高 PPV，必须提高特异度或限定高危人群。
  补充理由：本模型是在 155 例住院病人上开发并用同一批数据交叉验证的，没有外部验证；AUC 0.821 的 95% CI 很宽，换人群后排序能力本身也可能下降。
- 正确的定位要写出来：这个模型适合**高危人群的辅助分层**（如肝硬化监测、AFP 升高门诊），不适合社区初筛；若用于筛查，必须先在前瞻性高危队列中外部验证，并报告校准度而不只是 AUC。

**常见错误**

- 直接说"模型 PPV 41.4%，可以筛查"。41.4% 是 20.65% 患病率下的数字，换人群就崩。
- 用"灵敏度 × 患病率 / 特异度"之类的错误公式。分母必须是 `sens×prev + (1-spec)×(1-prev)`。
- 把 NPV 高（99.99%）当成"模型很好"。低患病率下任何检测的 NPV 都接近 100%，NPV 高不代表阳性结果有用。
- DeLong 只报 p 不报 z 和两个 AUC，或者把 ΔAUC 0.188 说成"提高了 18.8%"。AUC 是概率，差值 0.188 是绝对值，不是相对提升。
- DCA 只画曲线不报区间。验收要求"报告阈值范围和净收益"，必须给出数字（0.05-0.58）和至少两个阈值点的净收益。
- 净收益公式写错：`NB = TP/N - FP/N × pt/(1-pt)`，其中 TP/N 和 FP/N 都是相对**总样本**的比例，不是相对阳性人数。
- 阈值在测试集上选完又用同一测试集评估。Youden 阈值应在训练集或独立验证集上定。
- 用准确率评价模型。155 例里 123 例存活，全判阴性准确率 79.4%，比很多模型都高，但没有临床价值。
- 忽略模型的校准。AUC 0.821 只说明排序好，PPV 的计算还依赖概率是否校准；若模型过度自信，按阈值 0.127 分出的 2x2 表本身就有偏。

```python
from scipy import stats

# ---------- 1. 两个率的比较 ----------
p1, p2 = 0.25, 0.15          # 对照组 25%，干预组 15%
alpha, power = 0.05, 0.80
```

- 手算必须写出公式和每一步代入，不能只给结果。完整过程：

```text
z(1-0.05/2) = 1.9600
```

- Fleiss 连续性校正的结果 270 要一并给出，并说明"保守估计，正式方案里取 270 或 295（含脱落）"。两种口径都写出来，审稿人不会挑。
- 模拟部分要说清参数：两组 AUC 0.80 与 0.83，转化为标准化效应量 d = sqrt(2) × Φ⁻¹(AUC)，得 1.1902 与 1.3494，差 0.1592；两个标志物相关 0.6；300 次重复。结果：n = 1200 时功效 0.733，n = 1600 时 0.850，**线性内插约 1400-1500 例（每组约 700-750 例）才能达到 80% 功效**。
- 这个结果要能解释"为什么 0.03 的 AUC 差需要这么大样本"：AUC 的标准误约 0.014，两个相关 AUC 之差的标准误约 0.012，0.03 的差只有 2.5 个标准误；要让功效到 80%，z 需要约 2.8，差距还要再缩小标准误，样本必须上千。
- 第 3 问是本练习的思辨重点：**事后功效（post-hoc power）是观测 p 值的单调函数，不提供任何新信息。** 本机演示：p = 0.05 对应功效恰好 0.50，p = 0.001 对应 0.908，p = 0.9 对应 0.033。也就是说，"功效不足"只是"p 不显著"的另一种说法。正确做法是：在设计阶段用预期效应量算样本量；分析后报告**效应量和 95% CI**，而不是事后功效。
- 还要指出：用 300 例做事后功效，等于用观察到的效应量反推，而观察到的效应量本身有偏（显著时才被发表），会系统性高估功效。

**常见错误**

- 公式里用单侧 z：α = 0.05 双侧应取 1.96，写成 1.645 会把样本量算小约 20%。
- p̄ 算错：用 p1 或 p2 代替合并率，或用 (p1+p2) 不除以 2。
- 把两组的样本量当成总数。公式给出的是**每组** n，总样本是 2n = 500（不含脱落）。
- 脱落用乘法（250 × 1.15 = 288）。必须是除法：250 / 0.85 = 295。
- 忘记向上取整。249.98 要取 250，不能四舍五入成 250 就完事（这里恰好一样，但 249.2 就必须取 250）。
- 用模拟的 1600 例直接说"需要 1600 例"。1600 时功效 0.85，超过 80%，应按 80% 对应的点内插或直接报"1200 到 1600 之间，约 1400"。
- 事后功效写成"本研究功效只有 0.35，所以阴性结果不可信"。这是用观察到的效应量反推，逻辑循环；应报告 CI，说明"数据与中等效应相容，但不能排除小效应"。
- 把统计功效和临床意义混为一谈。样本量够只能保证"能检出"，不保证"检出的是重要效应"。
- 忽略多重比较对样本量的影响。如果有 3 个主要终点，α 要按 Bonferroni 分，样本量会显著增加。

### 练习 13-G（偏倚与 SAP）

> 本题是文字题。下面给出 **选项 B（SAP）的可照抄骨架**，并附 **选项 A（PSM）的实测结果与三条局限**。SAP 里的数字要替换成你自己课题的参数。

```text
【统计分析计划（Statistical Analysis Plan, SAP）模板】

一、研究概况
  1. 研究设计：前瞻性/回顾性队列（或 RCT），单中心/多中心
  2. 研究人群：纳入标准、排除标准、招募时间窗
  3. 样本量：见第四节，写明公式、参数、脱落率
  4. 分析人群定义（三套，缺一不可）：
     - 意向性分析集（intention-to-treat, ITT）：随机/入组即纳入
     - 符合方案集（per-protocol, PP）：完成方案者
     - 安全性分析集：至少接受一次干预者
     主要分析用 ITT，PP 作敏感性分析。

二、终点定义（在看数据前写死）
  1. 主要终点（primary endpoint）：只能有一个
     - 明确指标、测量时点、判定标准、谁判定（盲法？）
     - 例：术后 90 天内全因死亡
  2. 次要终点（secondary endpoints）：列出全部，不事后增删
  3. 安全性终点：单独一个 family
  4. 亚组分析：预先指定，写明分组切点和理由，标注"探索性"

三、主要分析
  1. 主要终点的效应量：
     - 二分类：RR 或 OR + 95% CI（写明用哪个、为什么）
     - 生存：HR + 95% CI（Cox），并报告 KM 曲线和 number at risk
     - 连续：均值差或中位数差 + 95% CI
  2. 检验方法：写明具体检验及前提（如"两组比较用 Mann-Whitney U，
     因 Shapiro-Wilk 提示偏态"）
  3. 协变量调整：预先列出调整变量及理由（不能看数据后挑）
  4. 缺失数据：
     - 缺失比例描述
     - 主分析：多重插补（MICE，m=20，含结局变量）
     - 敏感性分析：完整病例分析 + 最差情况填补
     - 不得默认"缺失=无事件"
  5. 模型诊断：比例风险假设（Schoenfeld）、共线性（VIF）、
     校准（校准曲线 + Brier）

四、样本量
  公式：n = [z(1-α/2)·sqrt(2·p̄·(1-p̄)) + z(1-β)·sqrt(p1(1-p1)+p2(1-p2))]²
            / (p1-p2)²
  参数：p1=0.25，p2=0.15，α=0.05（双侧），power=0.80
  结果：每组 250 例，脱落 15% → 每组 295 例，共 590 例
  说明：样本量按主要终点计算；次要终点不单独计算功效。

五、多重比较
  1. 主要终点：单个检验，α=0.05，不校正
  2. 次要终点 family：BH 法控制 FDR，q<0.05
  3. 亚组：Bonferroni 或标注"探索性，不作推断"
  4. 校正方法在揭盲/分析前锁定，不得事后更换

六、敏感性分析与稳健性
  1. 不同分析人群（ITT vs PP）
  2. 不同缺失处理（MI vs 完整病例 vs 最差情况）
  3. 不同模型设定（是否调整协变量）
  4. 剔除极端值/中心效应
  5. 阴性对照（negative control）如适用

七、报告规范
  - 遵循 STROBE（观察性）或 CONSORT（RCT）
  - 所有效应量带 95% CI
  - p 值保留三位小数，<0.001 写 p<0.001
  - 预先注册号、SAP 版本号与日期

八、变更记录
  日期 | 版本 | 变更内容 | 变更原因 | 是否在看数据前
```

```text
完整病例 n=87，暴露 22，对照 65，事件 17
匹配前 SMD:  Age -0.041, Bilirubin -0.609, Albumin +0.744, Protime +0.059
```

**判分要点**

- SAP 的第一条硬标准：**主要终点只能有一个**。写"主要终点为死亡、肝移植、肝癌复发"直接不及格。多个终点必须降为次要终点，并进入多重比较的 family。
- 样本量必须有公式、参数和代入过程，且与第四节一致（250/组、295/组含脱落、共 590）。只写"根据既往文献，计划入组 300 例"不得分。
- 缺失处理必须预先写死，且要区分主分析和敏感性分析。默认把缺失当"无事件"是本数据里真实存在的坑（腹水、白蛋白都有缺失）。
- 多重比较必须写清 family 划分：主要终点单独、次要终点一个 family、安全性终点另一个 family。不能把所有检验塞进一个 Bonferroni。
- 必须包含"变更记录"和"注册版本号"。SAP 的价值在于"看数据前锁定"，没有版本和日期的 SAP 只是一份草稿。
- PSM 部分（选项 A）要报：匹配前后 SMD、匹配对数、粗 OR 和匹配 OR。本题：22 对，最大 |SMD| 从 0.744 降到 0.264，粗 OR 0.333（0.070-1.592），匹配 OR 0.633（0.095-4.218）。
- PSM 的三条局限必须点明：
  1. **只平衡已测混杂**：年龄、胆红素、白蛋白、凝血酶原时间之外的混杂（如 HBV DNA、肝功能 Child-Pugh 分级、社会经济状态）完全没有处理，未测混杂仍然存在。
  2. **样本损失与精度下降**：65 个对照只匹配出 22 对，匹配后 OR 的 CI 从 0.070-1.592 变成 0.095-4.218，区间宽度从 1.52 增到 4.12；匹配后的估计不显著，既不能证明有效也不能证明无效。
  3. **匹配后仍有残余不平衡**：年龄的 |SMD| 从 0.041 升到 0.264，超过 0.1 的常用阈值。说明小样本 1:1 贪心匹配不能保证平衡，必须逐项检查 SMD，不能假定"做了 PSM 就平衡了"。
  补充：PSM 只能用于观察性研究的混杂控制，不能替代随机化；匹配后的 OR 仍是关联，不是因果效应。
- 暴露编码必须先验证。本数据中 `Antivirals` 取值 1 的 24 人死亡率 8.3%，取值 2 的 131 人死亡率 22.9%；抗病毒治疗应降低死亡，故 **1 = 使用**。若照抄 codebook 或想当然，OR 会从 0.33 变成 3.00，结论完全反向。

**常见错误**

- SAP 写成"研究方案"的复述，没有可执行的统计规则（用哪个检验、缺失怎么补、family 怎么分）。
- 主要终点有多个，或者写"主要终点将在数据分析后确定"。这等于承认会 p-hacking。
- 样本量只给结果不给公式，或公式里的 p1、p2 没有出处。
- 缺失处理写"删除缺失值"却不说明删除比例和可能的选择偏倚。
- 多重比较写"本研究不做校正，因为样本量小"。样本量小不是不校正的理由。
- 亚组分析不预先指定切点，看完数据后按最优切点分组（如"白蛋白 3.2 以下"），这是典型的 data dredging。
- PSM 只报匹配后 OR，不报匹配前 SMD 和匹配对数，读者无法判断匹配质量。
- 认为 PSM 能解决未测混杂。PSM 的前提是"所有混杂都已测量且模型正确"，这在观察性研究里几乎不可能完全满足。
- 匹配后不检查 SMD，直接宣称"两组基线可比"。本题年龄 SMD 0.264 就是反例。
- 用 PSM 匹配后的样本再做普通的 logistic 回归却不考虑配对结构。匹配样本应使用条件 logistic 或聚类稳健标准误。
- 把"匹配后 OR 不显著"写成"治疗无效"。CI 0.095-4.218 与"强保护"和"有害"都相容，只能说不确定。
